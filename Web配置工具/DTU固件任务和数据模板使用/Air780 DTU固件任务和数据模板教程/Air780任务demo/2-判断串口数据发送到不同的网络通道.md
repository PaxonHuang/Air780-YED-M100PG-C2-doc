# 一、功能描述

--串口数据发送的头,根据不同的标记发送到不同的网络通道

--相反,如果有多个串口,网络数据也可以发送到不同的串口

--关键点是在于如何处理字符串,判断数据内容

# 二、代码

下面是判断接收串口数据头部，前4个字节是hex 0x11 0x11 0x11 0x11 发送到网络通道1,  "aaa" 发送到 网络通道2，其余的发送到串口1。

这个就是多个通道和串口的分发功能。比如采集传感器的数据，传感器1的需要发送给平台1，传感器2的数据要发给平台2，传感器的数据处理结果需要发送给本地屏幕显示。

```lua
function
	local taskname="userDataTask"
	log.info(taskname,"start")
	local nid1,nid2=1,2
	local uid,uartr,uarts=1,"",""
	UartStopProRecCh(1)
	while true do
		uartr=UartGetRecChAndDel(uid)
		if uartr and string.len(uartr) >4 then 
			log.info(taskname,"uartr data",uartr)
			if string.byte(uartr,1)==0x11 and string.byte(uartr,2)==0x11 and string.byte(uartr,3)==0x11  and string.byte(uartr,4)==0x11 then 
				if 1==PronetGetNetSta(nid1) then 
					PronetSetSendCh(nid1,uartr)
				end
			elseif string.sub(uartr,1,3) =="aaa" then 
				if 1==PronetGetNetSta(nid2) then 
					PronetSetSendCh(nid2,uartr)
				end
      elseif string.sub(uartr,1,3) =="bbb" then 
				if 1==PronetGetNetSta(nid2) then 
					 PronetSetSendCh(nid2,string.sub(uartr,4,-1))
				end
			else
				UartSetSendCh(uid,uartr)
			end 
		end 
		sys.wait(100)
	end
end
```
