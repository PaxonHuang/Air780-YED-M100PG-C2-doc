# 一、功能描述

设备自动周期上报状态到服务器。检测输入状态是否变化，如果变化了马上上报,如果没变化定时周期300秒上报。

# 二、代码

```lua
function 
	local taskname="userTask"
	log.info(taskname,"start")
	sys.wait(15000)
	local nid=1
	local netsta =0
	local in1os,in2os,in3os,in4os=nil,nil,nil,nil
	local needup=1
	local lastuptime=0
	
	while true do 
		local d ={}
		local in1s=PerGetDiById(1)
		local in2s =PerGetDiById(2)
		local in3s =PerGetDiById(3)
		local in4s =PerGetDiById(4)
		d.datetime=os.date("%Y-%m-%d %H:%M:%S")
		d.csq=mobile.csq()
		d.imei=mobile.imei()
		d.iccid = PerGetIccid()
		d.vbat = PerGetVbattV()
		d.aiv = PerGetAdcGatherValByAdcId(0)
		d.aii = PerGetAdcGatherValByAdcId(1)
		d.In1 = in1s
		d.In2 = in2s
		d.In3 = in3s
		d.In4 = in4s
		if in1s~=in1os then 
			in1os =in1s
			needup=1
		end 
		if in2s~=in2os then 
			in2os =in2s
			needup=1
		end 
		if in3s~=in3os then 
			in3os =in3s
			needup=1
		end 
		if in4s~=in4os then 
			in4os =in4s
			needup=1
		end 
		local nowtime = os.time()
		if nowtime-lastuptime > 300 then 
			lastuptime = nowtime
			needup=1
		end 
		
		if needup ==1 then 
			local updata = json.encode(d)
			netsta = PronetGetNetSta(nid)
			log.info(taskname,"updata",updata,"netsta",netsta)
			if updata and netsta ==1 then 
				PronetSetSendCh(nid,updata)
			end
			needup=0
		end 
		sys.wait(100)
	end 
end
```
